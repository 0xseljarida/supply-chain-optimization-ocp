version: '3.8'
services:
  ids-engine:
    build: ./engine
    ports:
      - "8080:8080"
    environment:
      - MODEL_PATH=/models/cnn_ids.h5
      - LOG_LEVEL=INFO
    volumes:
      - ./models:/models
    restart: always

  mongodb:
    image: mongo:7.0
    ports:
      - "27017:27017"
    volumes:
      - mongo-data:/data/db

  dashboard:
    build: ./dashboard
    ports:
      - "3000:3000"
    depends_on:
